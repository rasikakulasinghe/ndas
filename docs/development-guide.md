## Quick Start

```bash
# 1. Clone and enter project
cd "D:\Projects\Current Projects\NDAS - Project\NDAS"

# 2. Create and activate virtual environment (Windows)
python -m venv venv
venv\Scripts\activate

# 3. Install dependencies
pip install -r requirements.txt

# 4. Set up environment variables
cp ".env" ".env.local"  # or create .env from .env.README.md

# 5. Run migrations
python manage.py makemigrations
python manage.py migrate

# 6. Create superuser
python manage.py createsuperuser

# 7. Start development server
python manage.py runserver
# App available at http://127.0.0.1:8000/
```

## Adding a New Model

### 1. Define the model

```python
# my_app/models.py
from ndas.custom_codes.Custom_abstract_class import TimeStampedModel, UserTrackingMixin
from ndas.custom_codes.choice import MY_CHOICES  # if needed, add to choice.py first

class MyModel(TimeStampedModel, UserTrackingMixin):
    patient = models.ForeignKey(
        'patients.Patient',
        on_delete=models.CASCADE,
        related_name='my_records',
        db_index=True,
    )
    name = models.CharField(max_length=200, db_index=True)
    status = models.CharField(max_length=20, choices=MY_CHOICES, db_index=True)

    class Meta:
        ordering = ['-created_at']
        indexes = [
            models.Index(fields=['patient', '-created_at'], name='mymodel_patient_date_idx'),
        ]
```

**Checklist:**
- [ ] Inherit `TimeStampedModel, UserTrackingMixin`
- [ ] Add `db_index=True` to all filterable/searchable fields
- [ ] Define composite indexes in `Meta.indexes`
- [ ] Add choices to `ndas/custom_codes/choice.py` (never inline)
- [ ] Never set `abstract = True` unless intentional

### 2. Create migration

```bash
python manage.py makemigrations my_app
python manage.py migrate
```

### 3. Register in admin

```python
# my_app/admin.py
from django.contrib import admin
from .models import MyModel

@admin.register(MyModel)
class MyModelAdmin(admin.ModelAdmin):
    list_display = ['name', 'patient', 'status', 'created_at']
    list_filter = ['status']
    search_fields = ['name', 'patient__baby_name']
```

## Adding a New View

```python
# my_app/views.py
from django.shortcuts import render, redirect, get_object_or_404
from django.contrib.auth.decorators import login_required
from django.views.decorators.http import require_GET, require_http_methods
from django_ratelimit.decorators import ratelimit
from ndas.custom_codes.error_handlers import handle_view_errors
from ndas.custom_codes.delete_helpers import has_delete_permission, validate_can_delete

@login_required(login_url="user-login")
@require_http_methods(["GET", "POST"])
@ratelimit(key='user_or_ip', rate='10/m')
@handle_view_errors(redirect_url='home', error_message='Error processing request')
def my_view(request, pk):
    obj = get_object_or_404(MyModel, id=pk)
    related = RelatedModel.objects.filter(parent=obj).select_related('added_by')

    if request.method == 'POST':
        form = MyModelForm(request.POST, instance=obj)
        if form.is_valid():
            form.save()
            messages.success(request, 'Saved successfully')
            return redirect('my-view', pk=obj.pk)
    else:
        form = MyModelForm(instance=obj)

    return render(request, 'my_app/edit.html', {
        'obj': obj,
        'form': form,
        'related': related,
    })
```

**Checklist:**
- [ ] `@login_required(login_url="user-login")`
- [ ] `@require_http_methods(["GET", "POST"])` or `@require_GET`
- [ ] `@ratelimit(key='user_or_ip', rate='10/m')` (create/edit) or `5/m` (delete)
- [ ] `@handle_view_errors(...)` for error handling
- [ ] `get_object_or_404()` — NEVER `objects.get()`
- [ ] `select_related()`/`prefetch_related()` for related objects

## Delete Implementation

### View

```python
@login_required(login_url="user-login")
@require_http_methods(["POST"])
@ratelimit(key='user_or_ip', rate='5/m')
def my_delete(request, pk):
    entity = get_object_or_404(MyModel, id=pk)

    if not has_delete_permission(request.user, entity):
        messages.error(request, "You do not have permission to delete this record.")
        return redirect('my-view', pk=pk)

    result = validate_can_delete(entity)
    if not result['can_delete']:
        messages.error(request, result['reason'])
        return redirect('my-view', pk=pk)

    entity.delete()
    messages.success(request, "Record deleted successfully.")
    return redirect(get_redirect_url('MyModel'))
```

### Template (add to detail/list view)

```django
{% load delete_modal_tags %}
{% include 'src/partials/delete_confirmation_modal.html' %}
```

## Input Sanitization Guidelines

### Free-text fields
```python
from ndas.custom_codes.validators import sanitize_text_input

# In form clean() or view before save
value = sanitize_text_input(form.cleaned_data['notes'])
```

### Rich text fields (CKEditor)
```python
from ndas.custom_codes.sanitization import sanitize_html

# Before saving RichTextField content
content = sanitize_html(form.cleaned_data['description'])
```

### File uploads
```python
from ndas.custom_codes.validators import validate_attachment_file, sanitize_filename

# As field validator
attachment = models.FileField(validators=[validate_attachment_file])
```

### Search queries
```python
from ndas.custom_codes.sanitization import sanitize_search_query

query = sanitize_search_query(request.GET.get('q', ''))
results = MyModel.objects.filter(name__icontains=query)
```

## Common Query Patterns

```python
# Always use get_object_or_404
from django.shortcuts import get_object_or_404
obj = get_object_or_404(MyModel, id=pk)

# Optimize with select_related for FKs
obj = get_object_or_404(
    MyModel.objects.select_related('patient', 'added_by', 'last_edit_by'),
    id=pk
)

# Optimize with prefetch_related for M2M
assessment = get_object_or_404(
    GMAssessment.objects.prefetch_related('diagnosis'),
    id=pk
)

# Safe count (returns 0 if None)
from ndas.custom_codes.custom_methods import getCountZeroIfNone
count = getCountZeroIfNone(MyModel.objects.filter(patient=patient))

# Age calculation
from ndas.custom_codes.custom_methods import calculate_age_string
age = calculate_age_string(patient.dob_tob, timezone.now(), format_type="medical")
# → "2 years and 3 months"
```

## Code Standards

### Imports order
```python
# 1. Standard library
import os
from datetime import datetime

# 2. Django
from django.db import models
from django.shortcuts import render

# 3. Third-party
from django_ratelimit.decorators import ratelimit

# 4. Local (custom_codes first, then app-level)
from ndas.custom_codes.Custom_abstract_class import TimeStampedModel, UserTrackingMixin
from ndas.custom_codes.choice import MY_CHOICES
from .models import MyModel
```

### Never do these
```python
# ❌ Never: inline choices
status = models.CharField(choices=[('a', 'Active'), ('i', 'Inactive')])
# ✅ Instead: add to choice.py and import

# ❌ Never: objects.get()
obj = MyModel.objects.get(id=pk)  # raises DoesNotExist
# ✅ Instead:
obj = get_object_or_404(MyModel, id=pk)

# ❌ Never: direct delete in views
entity.delete()
# ✅ Instead: use delete_helpers.py pattern

# ❌ Never: change Bootstrap version or AdminLTE version
# It breaks the entire UI

# ❌ Never: store secrets in code
SECRET_KEY = "hardcoded-secret"
# ✅ Instead: use .env / os.environ.get()
```
